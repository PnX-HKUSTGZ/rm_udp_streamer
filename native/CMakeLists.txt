cmake_minimum_required(VERSION 3.22)
project(rm_video VERSION 0.1.0 LANGUAGES CXX)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/.gdignore" "")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(RM_BUILD_GODOT "Build the Godot extension" ON)
option(RM_BUILD_MEDIA "Build the FFmpeg decoder" ${RM_BUILD_GODOT})
find_package(Threads REQUIRED)
add_library(rm_transport STATIC src/transport/protocol.cpp src/transport/udp_receiver.cpp)
target_include_directories(rm_transport PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(rm_transport PUBLIC Threads::Threads)
if(WIN32)
    target_link_libraries(rm_transport PUBLIC ws2_32)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(rm_transport PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(RM_BUILD_MEDIA)
    if(NOT FFMPEG_ROOT)
        message(FATAL_ERROR "Set FFMPEG_ROOT to the locked dependency cache")
    endif()
    foreach(component avcodec avutil swscale)
        if(NOT EXISTS "${FFMPEG_ROOT}/lib/lib${component}.a")
            message(FATAL_ERROR "Missing locked FFmpeg static library: ${component}")
        endif()
        add_library(ffmpeg_${component} STATIC IMPORTED)
        set_target_properties(ffmpeg_${component} PROPERTIES IMPORTED_LOCATION "${FFMPEG_ROOT}/lib/lib${component}.a")
    endforeach()
    add_library(rm_media STATIC src/media/decoder.cpp src/media/device.cpp src/media/cpu_preparer.cpp src/media/hevc_config.cpp src/media/session.cpp)
    target_include_directories(rm_media PUBLIC "${FFMPEG_ROOT}/include")
    target_link_libraries(rm_media PUBLIC rm_transport ffmpeg_avcodec ffmpeg_swscale ffmpeg_avutil)
    if(WIN32)
        target_link_libraries(rm_media PUBLIC bcrypt dxguid)
    else()
        target_link_libraries(rm_media PUBLIC m -Wl,-Bstatic atomic -Wl,-Bdynamic dl)
        # The explicit VAAPI FFmpeg recipe records libva/DRM in libavutil.pc.
        # CUDA-only fixed caches keep their existing dependency requirements.
        if(NOT EXISTS "${FFMPEG_ROOT}/lib/pkgconfig/libavutil.pc")
            message(FATAL_ERROR "Missing FFmpeg link metadata: lib/pkgconfig/libavutil.pc")
        endif()
        file(READ "${FFMPEG_ROOT}/lib/pkgconfig/libavutil.pc" RM_AVUTIL_LINK_METADATA)
        if(RM_AVUTIL_LINK_METADATA MATCHES "-lva")
            find_package(PkgConfig REQUIRED)
            pkg_check_modules(RM_VAAPI REQUIRED IMPORTED_TARGET libva libva-drm libdrm)
            target_link_libraries(rm_media PUBLIC PkgConfig::RM_VAAPI)
        endif()
    endif()
endif()
if(RM_BUILD_GODOT)
    if(NOT GODOT_CPP_DIR OR NOT RM_BUILD_MEDIA)
        message(FATAL_ERROR "GODOT_CPP_DIR and RM_BUILD_MEDIA are required")
    endif()
    set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
    set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE PATH "" FORCE)
    set(GODOTCPP_USE_HOT_RELOAD OFF CACHE BOOL "" FORCE)
    set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "" FORCE)
    add_subdirectory("${GODOT_CPP_DIR}" godot-cpp)
    add_library(rm_video SHARED src/godot/video_session.cpp src/godot/video_surface.cpp src/godot/cuda_gl_bridge.cpp)
    target_link_libraries(rm_video PRIVATE rm_media godot::cpp)
    set_target_properties(rm_video PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../addons/rm_video/bin"
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../addons/rm_video/bin")
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_link_options(rm_video PRIVATE -static-libgcc -static-libstdc++)
    endif()
    if(WIN32)
        set_target_properties(rm_video PROPERTIES PREFIX "")
        target_link_options(rm_video PRIVATE -Wl,--no-undefined -static -s)
    else()
        target_link_options(rm_video PRIVATE -Wl,--no-undefined -Wl,--exclude-libs,ALL -s)
    endif()
endif()
